--- description
There are `numCourses` courses labeled `0` to `numCourses - 1`. Each pair `prerequisites[i] = [a, b]` means course `b` must be taken before course `a`.

Return an order in which you can take all the courses. If several orders work, any of them is accepted. If it is impossible, return an empty array.

**Constraints**

--- tests
[2, [[1,0]]] => [0,1]
[4, [[1,0],[2,0],[3,1],[3,2]]] => [0,1,2,3]
[1, []] => [0]

[3, []]

--- python
class Solution:
    def findOrder(self, numCourses, prerequisites):
        g = [[] for _ in range(numCourses)]; indeg = [0] * numCourses
        for a, b in prerequisites:
            g[b].append(a); indeg[a] += 1
        q = deque(i for i in range(numCourses) if indeg[i] == 0); out = []
        while q:
            u = q.popleft(); out.append(u)
            for v in g[u]:
                indeg[v] -= 1
                if indeg[v] == 0:
                    q.append(v)
        return out if len(out) == numCourses else []

var findOrder = function(numCourses, prerequisites) {
    const g = Array.from({ length: numCourses }, () => []), state = new Array(numCourses).fill(0), out = [];
    for (const [a, b] of prerequisites) g[a].push(b);
    const dfs = u => { if (state[u] === 1) return false; if (state[u] === 2) return true; state[u] = 1; for (const v of g[u]) if (!dfs(v)) return false; state[u] = 2; out.push(u); return true; };
    for (let i = numCourses - 1; i >= 0; i--) if (!dfs(i)) return [];
    return out;

**Constraints**
- `1 <= nums.length <= 5000`
- `-10^4 <= nums[i], target <= 10^4`
--- tests
[[4,5,6,7,0,1,2], 0] => 4
[[4,5,6,7,0,1,2], 3] => -1
[[1], 0] => -1
[[1], 1]

[[1,2,3,4,5,6], 4]

class Solution:
    def search(self, nums, target):
        lo, hi = 0, len(nums) - 1
        while lo <= hi:
            mid = (lo + hi) // 2
            if nums[mid] == target:
                return mid
            if nums[lo] <= nums[mid]:
                if nums[lo] <= target < nums[mid]:
                    hi = mid - 1
                else:
                    lo = mid + 1
            else:
                if nums[mid] < target <= nums[hi]:
                    lo = mid + 1
                else:
                    hi = mid - 1
        return -1

class Trie:
    def __init__(self):
        self.root = {}
    def insert(self, word):
        n = self.root
        for c in word:
            n = n.setdefault(c, {})
        n['$'] = True

        n = self.root
        for c in s:
            if c not in n:

            n = n[c]
        return n

        n = self._walk(word)
        return n is not None and '$' in n

        return self._walk(prefix) is not None

class Solution:
    def isOneEditDistance(self, s, t):
        if len(s) > len(t):
            s, t = t, s
        if len(t) - len(s) > 1:
            return False
        for i in range(len(s)):
            if s[i] != t[i]:
                return s[i + 1:] == t[i + 1:] if len(s) == len(t) else s[i:] == t[i + 1:]
        return len(t) - len(s) == 1

var isOneEditDistance = function(s, t) {
    const m = s.length, n = t.length;
    if (Math.abs(m - n) > 1) return false;
    const dp = Array.from({ length: m + 1 }, (_, i) => Array.from({ length: n + 1 }, (_, j) => i === 0 ? j : j === 0 ? i : 0));

            if rem == 0:
                out.append(path[:]); return
            if i == len(candidates) or rem < 0:
                return
            path.append(candidates[i]); go(i, rem - candidates[i], path); path.pop()
            go(i + 1, rem, path)
        go(0, target, [])
        return out
--- javascript
var combinationSum = function(candidates, target) {
    const dp = Array.from({ length: target + 1 }, () => []);
    dp[0] = [[]];
    for (const c of candidates)
        for (let s = c; s <= target; s++)
            for (const comb of dp[s - c]) dp[s].push(comb.concat([c]));
    return dp[target];
};

Given the `root` of a binary tree, return `true` if it is a valid **binary search tree**:

- every value in a node's left subtree is **strictly less** than the node's value,
- every value in its right subtree is **strictly greater**, and

class Solution:
    def isValidBST(self, root):
        def ok(n, lo, hi):
            if not n:
                return True
            return lo < n.val < hi and ok(n.left, lo, n.val) and ok(n.right, n.val, hi)
        return ok(root, float('-inf'), float('inf'))
--- javascript
var isValidBST = function(root) {
    const vals = [];
    const walk = n => { if (!n) return; walk(n.left); vals.push(n.val); walk(n.right); };
    walk(root);
    for (let i = 1; i < vals.length; i++) if (vals[i] <= vals[i - 1]) return false;
    return true;

=== lowest-common-ancestor-of-a-binary-tree
title: Lowest Common Ancestor of a Binary Tree
difficulty: medium

--- description
Given a binary tree and two of its nodes `p` and `q`, return their **lowest common ancestor**: the deepest node that has both `p` and `q` as descendants. A node counts as a descendant of itself.

In the tests, `p` and `q` are given by value, and the judge passes you the actual node objects. The output shows the value of the node you return.

**Constraints**

- All values are unique, `p != q`, and both exist in the tree.
--- tests
[[3,5,1,6,2,0,8,null,null,7,4], 5, 1] => 3
[[3,5,1,6,2,0,8,null,null,7,4], 5, 4] => 5

[[3,5,1,6,2,0,8,null,null,7,4], 7, 6]
[[3,5,1,6,2,0,8,null,null,7,4], 0, 8]
[[1,2,3,null,4], 4, 3]
--- python
class Solution:
    def lowestCommonAncestor(self, root, p, q):
        if root is None or root is p or root is q:
            return root
        l = self.lowestCommonAncestor(root.left, p, q)
        r = self.lowestCommonAncestor(root.right, p, q)
        return root if l and r else l or r
--- javascript
var lowestCommonAncestor = function(root, p, q) {
    const parent = new Map([[root, null]]), st = [root];
    while (st.length) { const n = st.pop(); for (const c of [n.left, n.right]) if (c) { parent.set(c, n); st.push(c); } }
    const anc = new Set();
    for (let x = p; x; x = parent.get(x)) anc.add(x);
    let y = q;
    while (!anc.has(y)) y = parent.get(y);
    return y;
};

class Solution:
    def dailyTemperatures(self, temperatures):
        ans, st = [0] * len(temperatures), []
        for i, t in enumerate(temperatures):
            while st and temperatures[st[-1]] < t:
                j = st.pop(); ans[j] = i - j
            st.append(i)
        return ans
